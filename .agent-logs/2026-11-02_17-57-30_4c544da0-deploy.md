   ```bash
   sudo mkdir -p /opt/meetflow
   sudo chown "$(id -un):$(id -gn)" /opt/meetflow
   sudo usermod -aG docker "$(id -un)"

   openssl rand -hex 32
   openssl rand -hex 32
   ```
